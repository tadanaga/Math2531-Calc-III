\section{From the plane to space}
A point in the plane needs two coordinates. A point in space needs a third
coordinate to describe its height.

\begin{definition}[Coordinates in space]
An ordered triple $P=(x,y,z)$ locates a point in $\mathbb{R}^3$ relative to
three mutually perpendicular axes. The origin is $O=(0,0,0)$.
\end{definition}

\subsection*{Coordinate planes}
Setting one coordinate equal to zero leaves a plane:
\begin{center}
\begin{tabularx}{0.88\linewidth}{@{}XXX@{}}
\toprule
\textbf{$xy$-plane} & \textbf{$xz$-plane} & \textbf{$yz$-plane}\\
$z=0$ & $y=0$ & $x=0$\\
\bottomrule
\end{tabularx}
\end{center}

\begin{checkpoint}[Pause and sketch]
For $Q=(-2,3,-1)$, find its projection onto the $xz$-plane.
Which coordinate changes?
\answer{The projection is $(-2,0,-1)$; only the $y$-coordinate changes.}
\end{checkpoint}

\begin{commonmistake}
In $\mathbb{R}^3$, $x=2$ describes an entire \textbf{plane} parallel to the
$yz$-plane. Both $y$ and $z$ are free to vary.
\end{commonmistake}

\section{Distance and spheres}
The Pythagorean theorem extends naturally from two coordinates to three.

\begin{keyidea}[Distance in space]
For $P=(x_1,y_1,z_1)$ and $Q=(x_2,y_2,z_2)$,
\[
  d(P,Q)=\sqrt{(x_2-x_1)^2+(y_2-y_1)^2+(z_2-z_1)^2}.
\]
\end{keyidea}

% Original example syntax is preserved; the solution is its environment body.
\begin{example}{Find the distance between $P=(1,-2,3)$ and $Q=(4,2,3)$.}
\begin{align*}
  d(P,Q)&=\sqrt{(4-1)^2+(2+2)^2+(3-3)^2}\\
        &=\sqrt{9+16}=5.
\end{align*}
Both points have height $z=3$, so their separation lies entirely in a
horizontal plane.
\end{example}

\begin{definition}[Sphere]
A sphere is the set of points at a fixed distance $r>0$ from a center
$C=(a,b,c)$. Its equation is
\[
  (x-a)^2+(y-b)^2+(z-c)^2=r^2.
\]
\end{definition}

\begin{example}{Describe the sphere
  $x^2+y^2+z^2-4x+2y-6z+5=0$.}
Complete the square in each coordinate:
\begin{align*}
 (x^2-4x)+(y^2+2y)+(z^2-6z)&=-5,\\
 (x-2)^2+(y+1)^2+(z-3)^2&=9.
\end{align*}
The center is $\boxed{(2,-1,3)}$ and the radius is $\boxed{3}$.
\end{example}

\begin{checkpoint}[Your turn]
\begin{enumerate}
  \item Write the equation of the sphere centered at $(0,1,-2)$ with radius $4$.
  \item Describe the set $x^2+y^2+z^2\leq 9$.
\end{enumerate}
\answer{(1) $x^2+(y-1)^2+(z+2)^2=16$.
  (2) The closed ball of radius $3$ centered at the origin.}
\end{checkpoint}

\begin{takeaway}
Read an equation geometrically: a fixed coordinate gives a plane; a fixed
distance from a point gives a sphere.
\end{takeaway}
